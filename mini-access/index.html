Мини-Access

Настольное приложение на Windows Forms поверх MySQL: список таблиц, создание таблиц, добавление и удаление строк, генерация HTML-формы по схеме таблицы с живым предпросмотром в WebView2.

C#.NET 8Windows FormsMySQLMySql.DataWebView2
Скачать проект .zip Исходники на GitHub ↗

$ cat setup.txt

1. база

Запусти MySQL, в phpMyAdmin на вкладке SQL выполни:

CREATE DATABASE IF NOT EXISTS mini_access
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

2. проект

Распакуй архив и открой MiniAccess.csproj в Visual Studio. Пакеты NuGet подтянутся сами. Или скопируй файлы ниже в свой проект MiniAccess.

3. запуск

F5 → «Создать таблицу» students: id INT AUTO_INCREMENT + PK, name VARCHAR(100), age INT. При «Access denied» поправь Pwd= в Db.cs.

$ ls src/

Шаг 3. Подключение к MySQL и все запросы. Пароль MySQL меняется в ConnStr.

using System;
using System.Collections.Generic;
using System.Data;
using MySql.Data.MySqlClient;

public static class Db
{
    // XAMPP/OpenServer: пользователь root, пароль обычно пустой.
    // Если у вас другой пароль — впишите его после Pwd=
    const string ConnStr =
        "Server=localhost;Port=3306;Database=mini_access;Uid=root;Pwd=;CharSet=utf8mb4;";

    static MySqlConnection Open()
    {
        var c = new MySqlConnection(ConnStr);
        c.Open();
        return c;
    }

    public static List<string> ListTables()
    {
        var list = new List<string>();
        using var con = Open();
        var cmd = new MySqlCommand("SHOW TABLES", con);
        using var r = cmd.ExecuteReader();
        while (r.Read()) list.Add(r.GetString(0));
        return list;
    }

    public class Column { public string Name, Type; public bool IsPk; }

    public static List<Column> GetSchema(string table)
    {
        var cols = new List<Column>();
        using var con = Open();
        var cmd = new MySqlCommand("SHOW COLUMNS FROM `" + table + "`", con);
        using var r = cmd.ExecuteReader();
        while (r.Read())
            cols.Add(new Column
            {
                Name = r.GetString("Field"),
                Type = r.GetString("Type"),
                IsPk = r.GetString("Key") == "PRI"
            });
        return cols;
    }

    public static string GetPk(string table)
    {
        foreach (var c in GetSchema(table)) if (c.IsPk) return c.Name;
        return null;
    }

    public static void CreateTable(string name, List<Column> cols)
    {
        var defs = new List<string>();
        string pkName = null;
        foreach (var c in cols)
        {
            defs.Add("`" + c.Name + "` " + c.Type);
            if (c.IsPk) pkName = c.Name;
        }
        if (pkName != null) defs.Add("PRIMARY KEY(`" + pkName + "`)");
        using var con = Open();
        var cmd = new MySqlCommand(
            "CREATE TABLE `" + name + "` (" + string.Join(", ", defs) + ")", con);
        cmd.ExecuteNonQuery();
    }

    public static DataTable GetRows(string table)
    {
        var dt = new DataTable();
        using var con = Open();
        var da = new MySqlDataAdapter("SELECT * FROM `" + table + "`", con);
        da.Fill(dt);
        return dt;
    }

    public static void InsertRow(string table, Dictionary<string, string> values)
    {
        using var con = Open();
        var keys = new List<string>(values.Keys);
        var names = new List<string>();
        var parms = new List<string>();
        for (int i = 0; i < keys.Count; i++)
        {
            names.Add("`" + keys[i] + "`");
            parms.Add("@p" + i);
        }
        var cmd = new MySqlCommand(
            "INSERT INTO `" + table + "` (" + string.Join(",", names) + ") " +
            "VALUES (" + string.Join(",", parms) + ")", con);
        for (int i = 0; i < keys.Count; i++)
            cmd.Parameters.AddWithValue("@p" + i,
                string.IsNullOrEmpty(values[keys[i]]) ? (object)DBNull.Value : values[keys[i]]);
        cmd.ExecuteNonQuery();
    }

    public static void DeleteRow(string table, string pkCol, object pkValue)
    {
        using var con = Open();
        var cmd = new MySqlCommand(
            "DELETE FROM `" + table + "` WHERE `" + pkCol + "` = @pk", con);
        cmd.Parameters.AddWithValue("@pk", pkValue);
        cmd.ExecuteNonQuery();
    }
}

using System;
using System.Collections.Generic;
using System.Text.RegularExpressions;
using System.Windows.Forms;

public class CreateTableForm : Form
{
    public string TableName { get; private set; }
    public List<Db.Column> Columns { get; } = new List<Db.Column>();

    static readonly Regex NameRx = new Regex("^[A-Za-z_][A-Za-z0-9_]*$");

    public CreateTableForm()
    {
        Text = "Создать таблицу"; Width = 520; Height = 440;
        StartPosition = FormStartPosition.CenterParent;
        FormBorderStyle = FormBorderStyle.FixedDialog; MaximizeBox = false; MinimizeBox = false;

        var lbl = new Label { Text = "Имя таблицы:", Top = 10, Left = 10, Width = 200 };
        var txt = new TextBox { Top = 30, Left = 10, Width = 480 };

        var grid = new DataGridView
        {
            Top = 70, Left = 10, Width = 480, Height = 270,
            AllowUserToAddRows = true,
            AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill
        };
        grid.Columns.Add(new DataGridViewTextBoxColumn { HeaderText = "Имя колонки", Name = "name" });
        var typeCol = new DataGridViewComboBoxColumn { HeaderText = "Тип", Name = "type" };
        typeCol.Items.AddRange(
            "INT AUTO_INCREMENT", "INT", "VARCHAR(100)", "VARCHAR(255)",
            "TEXT", "DECIMAL(10,2)", "DATE", "DATETIME");
        grid.Columns.Add(typeCol);
        grid.Columns.Add(new DataGridViewCheckBoxColumn { HeaderText = "PK", Name = "pk", FillWeight = 30 });

        var btnOk = new Button { Text = "OK", Top = 355, Left = 300, Width = 90 };
        var btnNo = new Button { Text = "Отмена", Top = 355, Left = 400, Width = 90, DialogResult = DialogResult.Cancel };

        btnOk.Click += (s, e) =>
        {
            grid.EndEdit();
            TableName = txt.Text.Trim();
            if (!NameRx.IsMatch(TableName))
            {
                MessageBox.Show("Имя таблицы: только латиница, цифры и _, не с цифры.");
                return;
            }

            Columns.Clear();
            int pkCount = 0;
            foreach (DataGridViewRow row in grid.Rows)
            {
                if (row.IsNewRow) continue;
                string name = row.Cells["name"].Value?.ToString()?.Trim();
                string type = row.Cells["type"].Value?.ToString();
                bool pk = row.Cells["pk"].Value is bool b && b;
                if (string.IsNullOrEmpty(name) && string.IsNullOrEmpty(type)) continue;

                if (string.IsNullOrEmpty(name) || !NameRx.IsMatch(name))
                { MessageBox.Show("Неверное имя колонки: " + name); return; }
                if (string.IsNullOrEmpty(type))
                { MessageBox.Show("Не выбран тип у колонки " + name); return; }

                if (pk) pkCount++;
                Columns.Add(new Db.Column { Name = name, Type = type, IsPk = pk });
            }

            if (Columns.Count == 0) { MessageBox.Show("Добавьте хотя бы одну колонку."); return; }
            if (pkCount > 1) { MessageBox.Show("PK может быть только у одной колонки."); return; }

            DialogResult = DialogResult.OK; // закрывает диалог
        };

        Controls.Add(lbl); Controls.Add(txt); Controls.Add(grid);
        Controls.Add(btnOk); Controls.Add(btnNo);
        AcceptButton = btnOk; CancelButton = btnNo;
    }
}

using System.Collections.Generic;
using System.Windows.Forms;

public class AddRowForm : Form
{
    public Dictionary<string, string> Values { get; } = new Dictionary<string, string>();

    public AddRowForm(string table, List<Db.Column> cols)
    {
        Text = "Добавить строку в " + table; Width = 400; Height = 440;
        StartPosition = FormStartPosition.CenterParent;
        FormBorderStyle = FormBorderStyle.FixedDialog; MaximizeBox = false; MinimizeBox = false;

        var grid = new DataGridView
        {
            Top = 10, Left = 10, Width = 360, Height = 330,
            AllowUserToAddRows = false,
            AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill
        };
        grid.Columns.Add(new DataGridViewTextBoxColumn { HeaderText = "Поле", Name = "f", ReadOnly = true });
        grid.Columns.Add(new DataGridViewTextBoxColumn { HeaderText = "Значение", Name = "v" });
        foreach (var c in cols) grid.Rows.Add(c.Name, "");

        var btn = new Button { Text = "OK", Top = 355, Left = 280, Width = 90, DialogResult = DialogResult.OK };
        btn.Click += (s, e) =>
        {
            grid.EndEdit(); // фиксируем последнюю редактируемую ячейку
            foreach (DataGridViewRow row in grid.Rows)
                Values[row.Cells["f"].Value?.ToString()] = row.Cells["v"].Value?.ToString() ?? "";
        };

        Controls.Add(grid); Controls.Add(btn); AcceptButton = btn;
    }
}

using System;
using System.Collections.Generic;
using System.Data;
using System.IO;
using System.Linq;
using System.Text;
using System.Windows.Forms;

namespace MiniAccess
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
            RefreshTables();
            Load += Form1_Load;
        }

        // Асинхронная инициализация WebView2 — обязательно до NavigateToString
        private async void Form1_Load(object sender, EventArgs e)
        {
            try
            {
                await webView.EnsureCoreWebView2Async(null);
                webView.NavigateToString(
                    "<html><body style='font-family:sans-serif;color:#888'>" +
                    "Выберите таблицу и нажмите «Сгенерировать»</body></html>");
            }
            catch (Exception ex)
            {
                MessageBox.Show(
                    "WebView2 не инициализировался: " + ex.Message +
                    "\n\nУстановите Microsoft Edge WebView2 Runtime:\n" +
                    "https://developer.microsoft.com/microsoft-edge/webview2/");
            }
        }

        // ===== Вкладка «Таблицы» =====

        private void RefreshTables()
        {
            List<string> tables;
            try { tables = Db.ListTables(); }
            catch (Exception ex)
            {
                MessageBox.Show("Нет подключения к MySQL: " + ex.Message +
                    "\n\nПроверьте, что MySQL запущен и ConnStr в Db.cs верный.");
                return;
            }
            cbTables.Items.Clear(); cbTables.Items.AddRange(tables.ToArray());
            cbTablesHtml.Items.Clear(); cbTablesHtml.Items.AddRange(tables.ToArray());
            if (cbTables.Items.Count > 0) cbTables.SelectedIndex = 0;
            if (cbTablesHtml.Items.Count > 0) cbTablesHtml.SelectedIndex = 0;
        }

        private void cbTables_SelectedIndexChanged(object sender, EventArgs e)
        {
            if (cbTables.SelectedItem != null) LoadTable(cbTables.SelectedItem.ToString());
        }

        private void LoadTable(string table)
        {
            try { dgvData.DataSource = Db.GetRows(table); }
            catch (Exception ex) { MessageBox.Show(ex.Message, "Ошибка"); }
            dgvData.ReadOnly = true;
            dgvData.AllowUserToAddRows = false;
            dgvData.AllowUserToDeleteRows = false;
        }

        private void btnRefresh_Click(object sender, EventArgs e) => RefreshTables();

        private void btnCreateTable_Click(object sender, EventArgs e)
        {
            using var f = new CreateTableForm();
            if (f.ShowDialog(this) != DialogResult.OK) return;
            try
            {
                Db.CreateTable(f.TableName, f.Columns); RefreshTables();
                cbTables.SelectedItem = f.TableName;
            }
            catch (Exception ex) { MessageBox.Show(ex.Message, "Ошибка"); }
        }

        private void btnAddRow_Click(object sender, EventArgs e)
        {
            if (cbTables.SelectedItem == null) return;
            var t = cbTables.SelectedItem.ToString();
            using var f = new AddRowForm(t, Db.GetSchema(t));
            if (f.ShowDialog(this) != DialogResult.OK) return;
            try { Db.InsertRow(t, f.Values); LoadTable(t); }
            catch (Exception ex) { MessageBox.Show(ex.Message, "Ошибка"); }
        }

        private void btnDeleteRow_Click(object sender, EventArgs e)
        {
            if (cbTables.SelectedItem == null || dgvData.CurrentRow == null) return;
            var t = cbTables.SelectedItem.ToString();
            var pk = Db.GetPk(t);
            if (pk == null) { MessageBox.Show("В таблице нет первичного ключа"); return; }
            if (MessageBox.Show("Удалить строку?", "Подтверждение",
                MessageBoxButtons.YesNo) != DialogResult.Yes) return;
            try
            {
                Db.DeleteRow(t, pk, dgvData.CurrentRow.Cells[pk].Value);
                LoadTable(t);
            }
            catch (Exception ex) { MessageBox.Show(ex.Message, "Ошибка"); }
        }

        // ===== Вкладка «HTML-форма» =====

        private void btnGenerate_Click(object sender, EventArgs e)
        {
            if (cbTablesHtml.SelectedItem == null) return;
            string html = BuildHtml(cbTablesHtml.SelectedItem.ToString());
            txtHtml.Text = html;
            // Показываем HTML-страницу в браузерном движке (не как текст!)
            if (webView.CoreWebView2 != null)
                webView.NavigateToString(html);
        }

        private string BuildHtml(string table)
        {
            var cols = Db.GetSchema(table);
            var sb = new StringBuilder();
            sb.AppendLine("<!DOCTYPE html>");
            sb.AppendLine("<html lang='ru'><head><meta charset='utf-8'>");
            sb.AppendLine("<title>Форма: " + table + "</title>");
            sb.AppendLine("<style>body{font-family:sans-serif;margin:2em}" +
                          "label{display:block;margin:.6em 0}input{width:300px;padding:.3em}" +
                          "button{padding:.5em 1em}</style>");
            sb.AppendLine("</head><body>");
            sb.AppendLine("<h1>Форма: " + table + "</h1>");
            sb.AppendLine("<form method='post' action='save.php'>");
            foreach (var c in cols)
            {
                string t = c.Type.ToLowerInvariant();
                string input =
                    t.Contains("int") || t.Contains("decimal")
                        ? "<input type='number' step='any' name='" + c.Name + "'>"
                        : t.Contains("date")
                            ? "<input type='date' name='" + c.Name + "'>"
                            : "<input type='text' name='" + c.Name + "'>";
                sb.AppendLine("<label>" + c.Name + "<br>" + input + "</label>");
            }
            sb.AppendLine("<button type='submit'>Отправить</button>");
            sb.AppendLine("</form></body></html>");
            return sb.ToString();
        }

        private void btnSaveHtml_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrEmpty(txtHtml.Text)) return;
            using var dlg = new SaveFileDialog
            {
                Filter = "HTML (*.html)|*.html",
                FileName = (cbTablesHtml.SelectedItem?.ToString() ?? "form") + ".html"
            };
            if (dlg.ShowDialog() != DialogResult.OK) return;
            File.WriteAllText(dlg.FileName, txtHtml.Text, Encoding.UTF8);
            MessageBox.Show("Сохранено");
        }
    }
}
